# Tareas programadas de Boulder en un servidor Linux (crontab -e del usuario boulder).
# Las rutas suponen la instalación en /opt/boulder; ver docs/despliegue.md.

# Respaldo diario de la base a las 2:00, con 14 días de retención.
0 2 * * *    /opt/boulder/scripts/mantenimiento/respaldo_bd.sh /var/backups/boulder

# Verificación de salud cada 5 minutos.
*/5 * * * *  /opt/boulder/scripts/mantenimiento/verificar_salud.sh http://localhost:8080

# Copia semanal del último respaldo fuera del servidor (domingo 3:00).
0 3 * * 0    rsync -a --delete /var/backups/boulder/ respaldo@almacen:/respaldos/boulder/

# Estadísticas del planificador de PostgreSQL al día (domingo 4:00).
0 4 * * 0    psql -U postgres -d boulder_db -c 'VACUUM (ANALYZE);'
